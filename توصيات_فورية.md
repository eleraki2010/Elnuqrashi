```dart
Future<void> _holdSale() async {
  if (cart.isEmpty) {
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(content: Text('الفاتورة فارغة')),
    );
    return;
  }

  try {
    final saleInvoice = SaleInvoice(
      id: Uuid().v4(),
      invoiceNumber: _generateInvoiceNumber(),
      customerId: selectedCustomerId, // اختياري
      cashierId: null,
      items: cart.toList(),
      subtotal: subtotal,
      discount: discount,
      tax: tax,
      total: total,
      paidAmount: 0,
      changeDue: 0,
      paymentMethod: 'held',
      status: 'held', // ← مهم!
      createdAt: DateTime.now(),
    );

    await ref.read(saleRepositoryProvider).createSaleInvoice(saleInvoice);

    // مسح العربة
    ref.read(posCartProvider.notifier).clear();
    
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text('تم حفظ الفاتورة المعلّقة برقم: ${saleInvoice.invoiceNumber}'),
        backgroundColor: Colors.orange,
      ),
    );

    setState(() {
      cart = [];
      subtotal = 0;
      total = 0;
    });
  } catch (e) {
    print('خطأ: $e');
  }
}

Future<void> _showHeldInvoices() async {
  try {
    final heldInvoices = await ref.read(saleRepositoryProvider)
        .getSaleInvoicesByStatus('held');

    if (!mounted) return;

    showDialog(
      context: context,
      builder: (context) => AlertDialog(
        title: Text('الفواتير المعلّقة'),
        content: SizedBox(
          width: double.maxFinite,
          child: ListView.builder(
            itemCount: heldInvoices.length,
            itemBuilder: (context, index) {
              final invoice = heldInvoices[index];
              return ListTile(
                title: Text('الفاتورة #${invoice.invoiceNumber}'),
                subtitle: Text('${invoice.items.length} منتج - ${invoice.total} ج.م'),
                onTap: () => _loadHeldInvoice(invoice),
              );
            },
          ),
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(context),
            child: Text('إغلاق'),
          ),
        ],
      ),
    );
  } catch (e) {
    print('خطأ: $e');
  }
}

Future<void> _loadHeldInvoice(SaleInvoice invoice) async {
  // تحميل منتجات الفاتورة في العربة
  setState(() {
    cart = invoice.items.toList();
    subtotal = invoice.subtotal;
    discount = invoice.discount;
    tax = invoice.tax;
    total = invoice.total;
    selectedInvoiceId = invoice.id; // لتحديثها لاحقاً
  });

  if (mounted) {
    Navigator.pop(context); // إغلاق الـ Dialog
  }
}
```

#### Step 3: عدّل الـ complete sale لتحديث الفاتورة المعلّقة
```dart
Future<void> _completeSale() async {
  if (cart.isEmpty) return;

  try {
    final totalAmount = total;
    final paidAmount = double.parse(paidController.text);

    if (paidAmount < totalAmount && paymentMethod != 'credit') {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('المبلغ غير كافي')),
      );
      return;
    }

    final saleInvoice = SaleInvoice(
      id: selectedInvoiceId ?? Uuid().v4(), // ← استخدم الفاتورة القديمة إن وجدت
      invoiceNumber: _generateInvoiceNumber(),
      customerId: selectedCustomerId,
      items: cart.toList(),
      subtotal: subtotal,
      discount: discount,
      tax: tax,
      total: totalAmount,
      paidAmount: paidAmount,
      changeDue: paidAmount - totalAmount,
      paymentMethod: paymentMethod,
      status: 'completed',
      createdAt: DateTime.now(),
    );

    // إنشاء أو تحديث الفاتورة
    if (selectedInvoiceId != null) {
      await ref.read(saleRepositoryProvider).updateSaleInvoice(saleInvoice);
    } else {
      await ref.read(saleRepositoryProvider).createSaleInvoice(saleInvoice);
    }

    // حفظ StockMovement لكل منتج
    for (var item in cart) {
      await ref.read(stockMovementProvider.notifier).addMovement(
        productId: item.productId,
        type: 'sale',
        quantityChange: -item.quantity,
        referenceId: saleInvoice.id,
      );
    }

    // مسح العربة
    ref.read(posCartProvider.notifier).clear();
    _resetForm();

    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text('✓ تم إتمام البيع بنجاح'),
        backgroundColor: Colors.green,
      ),
    );
  } catch (e) {
    print('خطأ: $e');
  }
}
```

### المهمة الثانية: شاشة الجرد (3 ساعات)

**الملف الجديد**: `lib/presentation/screens/inventory_count_screen.dart`

```dart
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import '../widgets/stat_card.dart';
import '../../core/constants/app_constants.dart';

class InventoryCountScreen extends ConsumerStatefulWidget {
  const InventoryCountScreen({Key? key}) : super(key: key);

  @override
  ConsumerState<InventoryCountScreen> createState() => _InventoryCountScreenState();
}

class _InventoryCountScreenState extends ConsumerState<InventoryCountScreen> {
  final Map<String, TextEditingController> controllers = {};
  
  @override
  void dispose() {
    for (var controller in controllers.values) {
      controller.dispose();
    }
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    final productsAsync = ref.watch(allProductsProvider);

    return Scaffold(
      appBar: AppBar(
        title: const Text('جرد المخزون'),
        backgroundColor: AppColors.PRIMARY_BLUE,
      ),
      body: productsAsync.when(
        loading: () => const Center(child: CircularProgressIndicator()),
        error: (err, stack) => Center(child: Text('خطأ: $err')),
        data: (products) {
          return Column(
            children: [
              Padding(
                padding: const EdgeInsets.all(16),
                child: Text(
                  'أدخل الكمية الفعلية لكل منتج',
                  style: Theme.of(context).textTheme.titleMedium,
                  textAlign: TextAlign.center,
                ),
              ),
              Expanded(
                child: ListView.builder(
                  itemCount: products.length,
                  itemBuilder: (context, index) {
                    final product = products[index];
                    controllers[product.id] ??= TextEditingController();

                    return Card(
                      margin: const EdgeInsets.symmetric(horizontal: 12, vertical: 6),
                      child: Padding(
                        padding: const EdgeInsets.all(12),
                        child: Column(
                          children: [
                            Row(
                              mainAxisAlignment: MainAxisAlignment.spaceBetween,
                              children: [
                                Expanded(
                                  child: Column(
                                    crossAxisAlignment: CrossAxisAlignment.start,
                                    children: [
                                      Text(
                                        product.name,
                                        style: const TextStyle(fontWeight: FontWeight.bold),
                                      ),
                                      Text(
                                        'مسجّل: ${product.quantity} ${product.unit}',
                                        style: const TextStyle(fontSize: 12, color: Colors.grey),
                                      ),
                                    ],
                                  ),
                                ),
                                SizedBox(
                                  width: 100,
                                  child: TextField(
                                    controller: controllers[product.id],
                                    keyboardType: TextInputType.number,
                                    textAlign: TextAlign.center,
                                    decoration: InputDecoration(
                                      hintText: 'الفعلي',
                                      border: OutlineInputBorder(
                                        borderRadius: BorderRadius.circular(8),
                                      ),
                                      isDense: true,
                                    ),
                                  ),
                                ),
                              ],
                            ),
                          ],
                        ),
                      ),
                    );
                  },
                ),
              ),
              Padding(
                padding: const EdgeInsets.all(16),
                child: ElevatedButton.icon(
                  onPressed: () => _submitCount(),
                  icon: const Icon(Icons.check),
                  label: const Text('حفظ الجرد'),
                  style: ElevatedButton.styleFrom(
                    backgroundColor: AppColors.SUCCESS_GREEN,
                    minimumSize: const Size(double.infinity, 48),
                  ),
                ),
              ),
            ],
          );
        },
      ),
    );
  }

  Future<void> _submitCount() async {
    try {
      int changesCount = 0;

      for (var productId in controllers.keys) {
        final actualText = controllers[productId]!.text.trim();
        
        if (actualText.isEmpty) continue;

        final actual = int.parse(actualText);
        final product = ref.read(productByIdProvider(productId)).value;

        if (product != null) {
          final difference = actual - product.quantity;

          if (difference != 0) {
            await ref.read(stockMovementProvider.notifier).addMovement(
              productId: productId,
              type: 'count',
              quantityChange: difference,
              note: 'جرد الكمية',
            );
            changesCount++;
          }
        }
      }

      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text('✓ تم حفظ الجرد ($changesCount تصحيح)'),
            backgroundColor: Colors.green,
          ),
        );
        
        Navigator.pop(context);
      }
    } catch (e) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('خطأ: $e'), backgroundColor: Colors.red),
        );
      }
    }
  }
}
```
